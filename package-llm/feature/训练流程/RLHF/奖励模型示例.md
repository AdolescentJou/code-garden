import json

# 大模型配置
REQUEST_URL = 'https://aigc.sankuai.com/v1/openai/native/chat/completions'

# 全局变量用于控制调用频率

def _wait_for_rate_limit():
    """确保调用间隔不少于最小间隔时间"""
    global _last_call_time
    current_time = time.time()
    elapsed = current_time - _last_call_time

    if elapsed < _min_interval:
        wait_time = _min_interval - elapsed
        print(f"等待 {wait_time:.1f} 秒以避免超过API调用频率限制（间隔{_min_interval}秒）...")
        time.sleep(wait_time)

    _last_call_time = time.time()

def compute_score(data_source, solution_str, ground_truth, extra_info=None) -> float:
    """
    使用qwen3大模型来评估解决方案的质量

    Args:
        data_source: 数据源
        solution_str: 解决方案字符串
        ground_truth: 标准答案
        extra_info: 额外信息

    Returns:
        float: 0-1之间的分数

    try:
        # 构建评估提示词
        prompt = f"""
请作为一个专业的评估专家，评估以下解决方案的质量。

数据源: {data_source}
解决方案: {solution_str}
标准答案: {ground_truth}
{f"额外信息: {extra_info}" if extra_info else ""}

请从以下维度评估：
1. 准确性：解决方案是否正确
2. 完整性：是否包含了所有必要的要素
3. 清晰性：表达是否清晰明了
4. 实用性：是否具有实际应用价值

请给出一个0-1之间的综合分数，其中：
- 0.9-1.0: 优秀
- 0.7-0.8: 良好
- 0.5-0.6: 中等
- 0.3-0.4: 较差
- 0.0-0.2: 很差

只返回数字分数，不需要解释。
"""

        headers = {'Authorization': f'Bearer {APP_ID}'}
        request_content = {
            'messages': [{'role': 'user', 'content': prompt}],
            'model': MODEL_NAME,
            'max_tokens': 10,
            'stream': False
        }

        # 获取API调用锁，确保严格串行
        global _request_counter
        with _api_lock:
            _request_counter += 1
            current_request = _request_counter
            print(f"🔒 API请求 #{current_request} 开始执行...")

            # 确保调用间隔
            _wait_for_rate_limit()

            # 调用大模型
            print(f"📡 API请求 #{current_request} 正在调用大模型...")
            response = requests.post(REQUEST_URL, json=request_content, headers=headers)
            print(f"✅ API请求 #{current_request} 完成，释放锁")

        if response.status_code == 200:
            result = response.json()
            if 'choices' in result and len(result['choices']) > 0:
                score_text = result['choices'][0]['message']['content'].strip()

                # 尝试解析分数
                try:
                    score = float(score_text)
                    # 确保分数在0-1范围内
                    print(f"分数1: {score}")
                    return max(0.0, min(1.0, score))
                except ValueError:
                    # 如果无法解析数字，尝试从文本中提取
                    import re
                    numbers = re.findall(r'0\.\d+|1\.0|0|1', score_text)
                    if numbers:
                        score = float(numbers[0])
                        print(f"分数2: {score}")
                        return max(0.0, min(1.0, score))
                    else:
                        print(f"无法解析分数: {score_text}")
                        return 0.5  # 默认中等分数
            else:
                print("大模型响应格式异常")
                return 0.5
        elif response.status_code == 429:
            print("⚠️ 调用频率过高，触发RPM限制！")
            try:
                error_detail = response.json()
                print(f"错误详情: {error_detail}")
            except:
                print(f"响应内容: {response.text}")
            return 0.5
        elif response.status_code == 403:
            print("❌ 访问被拒绝，可能是权限或频率限制问题")
            try:
                error_detail = response.json()
                print(f"错误详情: {error_detail}")
            except:
                print(f"响应内容: {response.text}")
            return 0.5
        else:
            print(f"❌ 请求失败，状态码: {response.status_code}")
            try:
                error_detail = response.json()
                print(f"错误详情: {error_detail}")
            except:
                print(f"响应内容: {response.text}")
            return 0.5

    except Exception as e:
        print(f"计算分数时发生错误: {e}")
        return 0.5  # 发生错误时返回中等分数

def compute_score_batch(prompts_data) -> list:
    """
    批量计算多个prompt的奖励分数，一次API调用处理多个

    Args:
        prompts_data: list of dict, 每个dict包含:
            - data_source: 数据源
            - solution_str: 解决方案字符串
            - ground_truth: 标准答案
            - extra_info: 额外信息(可选)

    Returns:
        list: 对应的分数列表
    """
    if not prompts_data:
        return []

    try:
        # 构建批量评估提示词
        batch_prompt = "请作为专业评估专家，批量评估以下解决方案的质量。\n\n"

        for i, data in enumerate(prompts_data, 1):
            batch_prompt += f"=== 解决方案 {i} ===\n"
            batch_prompt += f"数据源: {data.get('data_source', '')}\n"
            batch_prompt += f"解决方案: {data.get('solution_str', '')}\n"
            batch_prompt += f"标准答案: {data.get('ground_truth', '')}\n"
            if data.get('extra_info'):
                batch_prompt += f"额外信息: {data['extra_info']}\n"
            batch_prompt += "\n"

        batch_prompt += """
评估维度：
1. 准确性：解决方案是否正确
2. 完整性：是否包含必要要素
3. 清晰性：表达是否清晰
4. 实用性：是否具有应用价值

分数标准（0-1之间）：
- 0.9-1.0: 优秀
- 0.7-0.8: 良好
- 0.5-0.6: 中等
- 0.3-0.4: 较差
- 0.0-0.2: 很差

请按顺序返回每个解决方案的分数，格式：[0.8, 0.6, 0.9, ...]
只返回分数数组，不需要解释。
"""

        headers = {'Authorization': f'Bearer {APP_ID}'}
        request_content = {
            'messages': [{'role': 'user', 'content': batch_prompt}],
            'model': MODEL_NAME,
            'max_tokens': 100,  # 增加token数以支持批量返回
            'stream': False
        }

        # 获取API调用锁，确保与单独评估串行
        global _request_counter
        with _api_lock:
            _request_counter += 1
            current_request = _request_counter
            print(f"🔒 批量API请求 #{current_request} 开始执行（处理{len(prompts_data)}个prompt）...")

            # 确保调用间隔
            _wait_for_rate_limit()

            # 调用大模型
            print(f"📡 批量API请求 #{current_request} 正在调用大模型...")
            response = requests.post(REQUEST_URL, json=request_content, headers=headers)
            print(f"✅ 批量API请求 #{current_request} 完成，释放锁")

        if response.status_code == 200:
            result = response.json()
            if 'choices' in result and len(result['choices']) > 0:
                response_text = result['choices'][0]['message']['content'].strip()

                # 尝试解析批量分数
                try:
                    # 使用正则提取数组格式的分数
                    import re
                    array_match = re.search(r'\[([\d\.\s,]+)\]', response_text)
                    if array_match:
                        scores_str = array_match.group(1)
                        scores = [float(x.strip()) for x in scores_str.split(',')]
                        # 确保分数数量匹配且在合理范围内
                        if len(scores) == len(prompts_data):
                            return [max(0.0, min(1.0, score)) for score in scores]

                    # 如果数组格式失败，尝试提取单独的数字
                    numbers = re.findall(r'0\.\d+|1\.0|0|1', response_text)
                    if len(numbers) >= len(prompts_data):
                        scores = [float(num) for num in numbers[:len(prompts_data)]]
                        return [max(0.0, min(1.0, score)) for score in scores]

                except Exception as parse_error:
                    print(f"批量分数解析失败: {parse_error}")
                    print(f"原始响应: {response_text}")

                # 解析失败，使用默认分数
                print("批量评估解析失败，使用默认分数")
                return [0.5] * len(prompts_data)

        elif response.status_code == 429:
            print("⚠️ 批量评估频率过高，触发RPM限制！")
            return [0.5] * len(prompts_data)
        else:
            print(f"❌ 批量评估请求失败，状态码: {response.status_code}")
            return [0.5] * len(prompts_data)

    except Exception as e:
        print(f"批量计算分数时发生错误: {e}")
        return [0.5] * len(prompts_data)

def compute_scores_smart(prompts_data) -> list:
    """
    智能分数计算调度器：根据prompt数量自动选择最佳策略

    Args:
        prompts_data: list of dict, 格式同compute_score_batch

    Returns:
        list: 对应的分数列表
    """
    if not prompts_data:
        return []

    num_prompts = len(prompts_data)

    # 策略选择
    if num_prompts == 1:
        # 单个prompt：直接调用单独评估
        data = prompts_data[0]
        score = compute_score(
            data.get('data_source', ''),
            data.get('solution_str', ''),
            data.get('ground_truth', ''),
            data.get('extra_info')
        )
        return [score]

    elif num_prompts <= 10:
        # 10个以内：使用批量评估（推荐）
        print(f"📦 使用批量评估模式处理 {num_prompts} 个prompt")
        return compute_score_batch(prompts_data)

    else:
        # 超过10个：分批处理
        print(f"🔄 分批处理 {num_prompts} 个prompt（每批10个）")
        all_scores = []
        batch_size = 10

        for i in range(0, num_prompts, batch_size):
            batch = prompts_data[i:i + batch_size]
            batch_scores = compute_score_batch(batch)
            all_scores.extend(batch_scores)
            print(f"完成第 {i//batch_size + 1} 批次：{len(batch)} 个prompt")

        return all_scores

def set_rate_limit(interval_seconds):
    """
    动态调整API调用频率限制

    Args:
        interval_seconds: 调用间隔秒数
    """
    global _min_interval
    _min_interval = interval_seconds
    print(f"🔧 API调用间隔已调整为 {interval_seconds} 秒")

def get_api_status():
    """
    获取当前API调用状态

    Returns:
        dict: 包含状态信息的字典
    """
    global _request_counter, _api_lock

    # 尝试获取锁状态（非阻塞）
    is_locked = not _api_lock.acquire(blocking=False)
    if not is_locked:
        # 如果成功获取锁，立即释放
        _api_lock.release()

    status = {
        'total_requests': _request_counter,
        'is_api_busy': is_locked,
        'rate_limit_interval': _min_interval,
        'last_call_time': _last_call_time
    }

    return status

def print_api_status():
    """
    打印当前API调用状态
    """
    status = get_api_status()

    print("=" * 50)
    print("🔍 API调用状态监控")
    print("=" * 50)
    print(f"📊 总请求数: {status['total_requests']}")
    print(f"🚦 API状态: {'🔴 忙碌中' if status['is_api_busy'] else '🟢 空闲'}")
    print(f"⏱️ 频率限制: {status['rate_limit_interval']} 秒间隔")

    if status['last_call_time'] > 0:
        elapsed = time.time() - status['last_call_time']
        print(f"🕒 距上次调用: {elapsed:.1f} 秒前")
    else:
        print(f"🕒 距上次调用: 尚未调用")

    print("=" * 50)

def wait_for_api_idle(timeout=60):
    """
    等待API变为空闲状态

    Args:
        timeout: 超时时间（秒），默认60秒

    Returns:
        bool: True表示API变为空闲，False表示超时
    """
    start_time = time.time()

    while time.time() - start_time < timeout:
        if not get_api_status()['is_api_busy']:
            print("✅ API已变为空闲状态")
            return True

        print("⏳ 等待API空闲...")
        time.sleep(1)

    print("⏰ 等待API空闲超时")
    return False
