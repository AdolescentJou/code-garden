### 整体说明
- 无分布式、无混合精度、无梯度累积，单卡 / CPU 即可运行；
- 模型基于 Transformer 极简结构，仅保留核心的「嵌入层 + 单隐藏层 + 输出层」；
- 数据使用极简的 JSONL 格式，仅需准备少量文本即可验证；
- 配套极简 Tokenizer（基于字符级，无需额外下载预训练词表，降低入门门槛）；
- 所有代码整合为单个文件，无需额外创建子模块（如 model/dataset），方便调试。

```python
# 导入核心依赖，仅保留PyTorch基础库，无冗余依赖
import os
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import Dataset, DataLoader
import json
from typing import List

# ===================== 步骤1：定义全局配置（替代argparse，零基础更易上手） =====================
# 所有配置直接写死，无需命令行参数，后续阶段再替换为argparse
CONFIG = {
    "device": "cuda:0" if torch.cuda.is_available() else "cpu",  # 训练设备，自动检测GPU/CPU
    "batch_size": 4,  # 小批量，降低内存占用，确保能跑通
    "hidden_size": 128,  # 模型隐藏层维度，最小容量，训练最快
    "max_seq_len": 64,  # 文本最大截断长度，适配小模型输入
    "data_path": "small_pretrain.jsonl",  # 小型训练数据路径（需自己创建，格式见下文）
    "learning_rate": 5e-4,  # 初始学习率，大模型预训练主流值
    "epochs": 1  # 仅训练1轮，验证流程闭环即可
}

# ===================== 步骤2：实现极简Tokenizer（字符级，无需预训练词表） =====================
# 大模型Tokenizer核心作用：文本→数字（token_id）、数字→文本，这里实现最基础的字符级版本
# 字符级：将每个汉字/字母/符号视为一个token，适合中文小数据训练，无需复杂的词表构建
class SimpleCharTokenizer:
    def __init__(self):
        # 初始化词表：特殊符号（pad补0、unk未知字符）+ 空字典（后续从数据中构建）
        self.special_tokens = ["<pad>", "<unk>"]  # pad：补全长度，unk：未见过的字符
        self.token2id = {token: idx for idx, token in enumerate(self.special_tokens)}  # 符号→id
        self.id2token = {idx: token for token, idx in self.token2id.items()}  # id→符号
        self.pad_token = "<pad>"
        self.unk_token = "<unk>"
        self.pad_id = self.token2id[self.pad_token]
        self.unk_id = self.token2id[self.unk_token]

    # 从训练数据构建词表：遍历所有文本，收集所有字符，避免unk过多
    def build_vocab(self, texts: List[str]):
        for text in texts:
            for char in text:
                if char not in self.token2id:
                    self.token2id[char] = len(self.token2id)
                    self.id2token[len(self.id2token)] = char
        print(f"词表构建完成，总大小：{len(self.token2id)}（含{len(self.special_tokens)}个特殊符号）")

    # 文本转token_id：核心方法，训练时用
    def encode(self, text: str, max_len: int) -> List[int]:
        # 1. 文本转字符列表，过滤空字符
        chars = [c for c in text if c.strip()]
        # 2. 截断：超过max_len的部分直接舍弃，适配模型输入长度
        chars = chars[:max_len]
        # 3. 字符转id：未见过的字符用unk_id替代
        token_ids = [self.token2id.get(c, self.unk_id) for c in chars]
        # 4. 补全（pad）：不足max_len的部分用pad_id补到固定长度
        pad_len = max_len - len(token_ids)
        token_ids += [self.pad_id] * pad_len
        return token_ids

    # token_id转文本：验证时用，可选
    def decode(self, token_ids: List[int]) -> str:
        chars = [self.id2token.get(idx, self.unk_token) for idx in token_ids]
        # 去掉pad符号，拼接成文本
        return "".join([c for c in chars if c != self.pad_token])

# ===================== 步骤3：实现极简预训练数据集（PretrainDataset） =====================
# 数据集核心作用：加载数据文件→解析文本→调用Tokenizer转token_id→返回模型可直接输入的张量
# 预训练任务为「自回归语言建模」：输入文本的前n-1个token，预测第n个token，标签与输入一致（移位）
class PretrainDataset(Dataset):
    def __init__(self, data_path: str, tokenizer: SimpleCharTokenizer, max_seq_len: int):
        self.max_seq_len = max_seq_len
        self.tokenizer = tokenizer
        self.texts = self._load_data(data_path)  # 加载原始文本
        self.tokenizer.build_vocab(self.texts)   # 用原始文本构建词表
        self.input_ids = self._process_texts()   # 所有文本转token_id，预处理一次，训练时直接用

    # 加载JSONL数据：每行是{"text": "待训练文本"}格式，返回文本列表
    def _load_data(self, data_path: str) -> List[str]:
        if not os.path.exists(data_path):
            raise FileNotFoundError(f"数据文件不存在：{data_path}，请按格式创建（见代码说明）")
        texts = []
        with open(data_path, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                data = json.loads(line)
                if "text" in data and data["text"].strip():
                    texts.append(data["text"].strip())
        print(f"数据加载完成，共{len(texts)}条文本")
        return texts

    # 预处理所有文本：转token_id，生成模型输入
    def _process_texts(self) -> List[List[int]]:
        input_ids = []
        for text in self.texts:
            token_ids = self.tokenizer.encode(text, self.max_seq_len)
            input_ids.append(token_ids)
        return input_ids

    # 数据集核心方法：获取单个样本（索引对应）
    def __getitem__(self, idx: int) -> torch.Tensor:
        # 返回tensor类型，PyTorch模型仅支持张量输入
        return torch.tensor(self.input_ids[idx], dtype=torch.long)

    # 数据集核心方法：返回样本总数，DataLoader需要
    def __len__(self) -> int:
        return len(self.input_ids)

# ===================== 步骤4：定义极简MiniMind模型（Transformer核心结构） =====================
# 模型基于Transformer自回归架构，仅保留「嵌入层→单隐藏层→层归一化→输出层」
# 抛弃所有高级模块（如多头注意力、MoE、多层堆叠），仅保留能完成自回归任务的最小结构
# 预训练任务：自回归语言建模（LM），输入input_ids，预测下一个token，计算交叉熵损失
class MiniMind(nn.Module):
    def __init__(self, vocab_size: int, hidden_size: int, max_seq_len: int, pad_id: int):
        super().__init__()
        self.vocab_size = vocab_size  # 词表大小，输出层维度与词表一致
        self.hidden_size = hidden_size  # 隐藏层维度
        self.pad_id = pad_id  # pad_id，计算损失时忽略pad部分

        # 1. 嵌入层（Token Embedding）：将token_id（整数）转为稠密向量（hidden_size维度）
        # nn.Embedding：词嵌入层，输入[batch_size, max_seq_len]，输出[batch_size, max_seq_len, hidden_size]
        self.embedding = nn.Embedding(vocab_size, hidden_size, padding_idx=pad_id)

        # 2. 极简隐藏层：用1层全连接层替代Transformer的多头注意力+前馈网络，降低复杂度
        self.hidden_layer = nn.Linear(hidden_size, hidden_size)
        self.activation = nn.GELU()  # 激活函数，大模型主流选择（比Relu效果好）

        # 3. 层归一化：稳定训练，防止梯度爆炸/下溢，大模型必备
        self.layer_norm = nn.LayerNorm(hidden_size)

        # 4. 输出层：将隐藏层向量映射回词表维度，用于预测下一个token
        self.output_layer = nn.Linear(hidden_size, vocab_size)

        # 初始化参数：保证模型初始参数分布合理，训练易收敛
        self.apply(self._init_weights)

    # 模型参数初始化：大模型标准初始化方式
    def _init_weights(self, module):
        if isinstance(module, nn.Embedding):
            # 嵌入层参数初始化：正态分布，均值0，标准差0.02
            nn.init.normal_(module.weight, mean=0.0, std=0.02)
        elif isinstance(module, nn.Linear):
            # 全连接层参数初始化：正态分布+偏置置0
            nn.init.normal_(module.weight, mean=0.0, std=0.02)
            if module.bias is not None:
                nn.init.zeros_(module.bias)
        elif isinstance(module, nn.LayerNorm):
            # 层归一化参数初始化：权重置1，偏置置0
            nn.init.ones_(module.weight)
            nn.init.zeros_(module.bias)

    # 前向传播：模型核心逻辑，输入input_ids，输出预测logits和损失
    # input_ids：[batch_size, max_seq_len]，模型输入的token_id张量
    def forward(self, input_ids):
        # 步骤1：嵌入层转换，得到词向量
        # x: [batch_size, max_seq_len, hidden_size]
        x = self.embedding(input_ids)

        # 步骤2：隐藏层计算+激活+层归一化
        # x: [batch_size, max_seq_len, hidden_size]
        x = self.hidden_layer(x)
        x = self.activation(x)
        x = self.layer_norm(x)

        # 步骤3：输出层映射到词表维度，得到预测logits
        # logits: [batch_size, max_seq_len, vocab_size]，每个token对应词表中所有token的预测分数
        logits = self.output_layer(x)

        # 步骤4：计算自回归语言建模损失（交叉熵损失）
        # 自回归任务：输入[CLS, A, B, C]，预测[A, B, C, D]，因此标签是输入移位一位
        # 输入：input_ids[:, :-1]（前n-1个token），标签：input_ids[:, 1:]（后n-1个token）
        shift_logits = logits[:, :-1, :].contiguous()  # 去掉最后一个token的预测，[batch_size, max_seq_len-1, vocab_size]
        shift_labels = input_ids[:, 1:].contiguous()   # 去掉第一个token，作为标签，[batch_size, max_seq_len-1]

        # 计算交叉熵损失：忽略pad_id对应的标签，避免无效损失影响训练
        loss_fct = nn.CrossEntropyLoss(ignore_index=self.pad_id)
        # 交叉熵输入要求：logits展平为[batch_size*(max_seq_len-1), vocab_size]，labels展平为[batch_size*(max_seq_len-1)]
        loss = loss_fct(shift_logits.view(-1, self.vocab_size), shift_labels.view(-1))

        # 返回预测logits和损失，logits可选（训练时主要用loss，验证时用logits）
        return {"logits": logits, "loss": loss}

# ===================== 步骤5：实现极简日志打印（Logger） =====================
# 替代复杂的日志库，仅打印训练关键信息（step、loss），零基础易查看
def Logger(msg: str):
    """极简日志打印，带时间戳（可选）"""
    import time
    current_time = time.strftime("%Y-%m-%d %H:%M:%S", time.localtime())
    print(f"[{current_time}] {msg}")

# ===================== 步骤6：主训练逻辑（核心：跑通单次前向+反向传播） =====================
def main():
    # ===== 6.1 初始化设备，打印配置信息 =====
    device = torch.device(CONFIG["device"])
    Logger(f"训练设备：{device}")
    Logger(f"训练配置：{CONFIG}")

    # ===== 6.2 初始化Tokenizer、数据集、DataLoader =====
    # Tokenizer初始化
    tokenizer = SimpleCharTokenizer()
    # 数据集初始化：加载数据+构建词表+预处理
    dataset = PretrainDataset(
        data_path=CONFIG["data_path"],
        tokenizer=tokenizer,
        max_seq_len=CONFIG["max_seq_len"]
    )
    # DataLoader：数据加载器，将数据集按batch_size分组，训练时迭代取数
    # num_workers=0：主进程加载数据，避免多线程报错，零基础友好
    # pin_memory=False：关闭内存锁，降低内存占用
    dataloader = DataLoader(
        dataset,
        batch_size=CONFIG["batch_size"],
        shuffle=True,  # 训练时打乱数据，提升模型泛化能力
        num_workers=0,
        pin_memory=False
    )
    Logger(f"DataLoader初始化完成，共{len(dataloader)}个batch（每batch{CONFIG['batch_size']}条样本）")

    # ===== 6.3 初始化模型、优化器 =====
    # 模型初始化：传入词表大小、隐藏层维度、最大序列长度、pad_id
    model = MiniMind(
        vocab_size=len(tokenizer.token2id),
        hidden_size=CONFIG["hidden_size"],
        max_seq_len=CONFIG["max_seq_len"],
        pad_id=tokenizer.pad_id
    ).to(device)  # 将模型移到指定设备（GPU/CPU）
    Logger(f"模型初始化完成，总参数量：{sum(p.numel() for p in model.parameters()):,} 个")

    # 优化器初始化：AdamW，大模型预训练主流优化器（比Adam更适合预训练，带权重衰减）
    # weight_decay=0.01：权重衰减，防止过拟合
    optimizer = optim.AdamW(model.parameters(), lr=CONFIG["learning_rate"], weight_decay=0.01)

    # ===== 6.4 开始训练（核心循环） =====
    model.train()  # 将模型设为训练模式（影响Dropout、BatchNorm等层，本模型无，但养成习惯）
    total_steps = len(dataloader)  # 每轮的总step数（batch数）
    Logger("开始训练（阶段1：最小训练单元）")

    for epoch in range(CONFIG["epochs"]):
        epoch_loss = 0.0  # 记录本轮总损失
        for step, input_ids in enumerate(dataloader, start=1):
            # 步骤1：将数据移到指定设备，input_ids: [batch_size, max_seq_len]
            input_ids = input_ids.to(device)

            # 步骤2：前向传播，计算损失（核心：调用model.forward，得到loss）
            outputs = model(input_ids)
            loss = outputs["loss"]  # 获取当前batch的损失值

            # 步骤3：反向传播，计算梯度（核心：loss.backward()，PyTorch自动求导）
            loss.backward()

            # 步骤4：优化器更新参数（核心：optimizer.step()，用梯度更新模型参数）
            optimizer.step()

            # 步骤5：清空梯度（必须！否则下一个batch的梯度会累加，导致训练错误）
            optimizer.zero_grad(set_to_none=True)  # set_to_none=True：更高效的梯度清空方式

            # 累计本轮损失
            epoch_loss += loss.item()  # loss.item()：取出张量的数值，避免计算图占用内存

            # 打印日志：每1个step打印一次，阶段1无需间隔，看清楚每一步的损失变化
            Logger(f"Epoch: [{epoch+1}/{CONFIG['epochs']}] Step: [{step}/{total_steps}] Loss: {loss.item():.4f}")

        # 计算本轮平均损失
        avg_epoch_loss = epoch_loss / total_steps
        Logger(f"Epoch [{epoch+1}/{CONFIG['epochs']}] 训练完成，平均损失：{avg_epoch_loss:.4f}")

    # ===== 6.5 训练结束，简单验证模型参数更新 =====
    # 验证：打印模型第一层（嵌入层）的部分权重，确认参数已更新（可选，零基础可忽略）
    first_embedding_weight = model.embedding.weight.data[0:2, 0:5].cpu().numpy()
    Logger(f"训练完成，嵌入层前2行前5列权重（验证参数已更新）：\n{first_embedding_weight}")
    Logger("阶段1训练完成：成功跑通最小训练单元（前向+反向传播+参数更新）！")

# ===================== 步骤7：程序入口 =====================
if __name__ == "__main__":
    main()
```
